from javascript import require, On, globalThis

mineflayer = require('mineflayer')
bot = mineflayer.createBot({
    'host': '127.0.0.1',
    'username': 'chunk_loader',
    'version': "1.16.5"
})

@On(bot, 'chat')
def handle_chat(this, username, message, *args):
    if 'рыбачь' in message.lower():
        bot.chat(f'ок, {username}, иду нахуй!')
        globalThis.setTimeout(lambda: start_fishing(), 100)

@On(bot, 'playerCollect')
def on_collect(this, collector, item, *args):
    if collector.username == bot.username:
        if not fish_caught:
            fish_caught = True
            print(f"🎣 Рыба поймана!")
            globalThis.setTimeout(lambda: start_fishing(), 500)

@On(bot, 'soundEffectHeard')
def on_sound(this, sound_name, *args):
    if sound_name == 'entity.fishing_bobber.splash':
        print("🎣 Поклёвка! Подсекаем...")
        bot.activateItem()
